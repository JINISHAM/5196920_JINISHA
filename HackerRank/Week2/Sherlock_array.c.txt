#include <stdio.h>

void balancedSums(int n,int arr[]){
    int total_sum = 0,sum_left = 0;
    for(int i=0;i<n;i++){
        total_sum += arr[i];
    }
    
    for(int i=0;i<n;i++){
        int sum_right;
        sum_right = total_sum - sum_left - arr[i];
        if(sum_left == sum_right){
             printf("YES\n");
             return;
        }
        sum_left += arr[i];
    }
    printf("NO\n");
}

int main()
{
    int t;
    scanf("%d",&t);
    while(t--){
        int n;
        scanf("%d",&n);
        int arr[n];
        for(int i=0;i<n;i++){
            scanf("%d",&arr[i]);
        }
        balancedSums(n,arr);
    }
    
    return 0;
}
