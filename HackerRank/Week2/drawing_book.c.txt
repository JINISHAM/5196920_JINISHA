#include <stdio.h>
int minimum_pages(int n,int p){
    int front = p/2;
    int back = (n/2) - (p/2);
    if(front < back)
        return front;
    else
        return back;
}
int main(){
    int n,p;
    scanf("%d",&n);
    scanf("%d",&p);
    int minimum = minimum_pages(n,p);
    printf("%d",minimum);
    return 0;
}
