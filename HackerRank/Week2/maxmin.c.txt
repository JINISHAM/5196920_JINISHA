#include <stdio.h>
#include <stdlib.h>

int compare(const void *a,const void *b){
    return (*(int*)a - *(int*)b);
}

int maxmin(int n,int k,int arr[]){
    qsort(arr,n,sizeof(int),compare);
    int min_unfair = arr[k-1] - arr[0];
    for(int i=0;i<=n-k;i++){
        int unfairness = arr[i+k-1] - arr[i];
        if(unfairness < min_unfair)
            min_unfair = unfairness;
    }
    return min_unfair;
    
}

int main()
{
    int n;
    scanf("%d",&n);
    int k;
    scanf("%d",&k);
    int arr[n];
    for(int i=0;i<n;i++){
        scanf("%d",&arr[i]);
    }
    int result = maxmin(n,k,arr);
    printf("%d",result);
    return 0;
}
