#include <stdio.h>
#include <ctype.h>

void Caesar_Cipher(int n,char str[],int k){
    char cipher[n+1];
    for(int i=0;i<n;i++){
        if(isupper(str[i])){
            cipher[i] = 'A' + (str[i] - 'A' + k) % 26;
        }
        else if (islower(str[i])) {
            cipher[i] = 'a' + (str[i] - 'a' + k) % 26;
        }
        else{
            cipher[i] = str[i];
        } 
    }
    cipher[n] = '\0';
    printf("%s",cipher);

}
int main()
{
    int n;
    scanf("%d",&n);
    char str[n+1];
    scanf(" %[^\n]s",str);
    int k;
    scanf("%d",&k);
    Caesar_Cipher(n,str,k);
}
