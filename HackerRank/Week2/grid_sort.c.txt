#include <stdio.h>
#include <stdlib.h>

int compare(const void *a,const void *b){
    return (*(char*)a - *(char*)b);
}

void grid(int n,char str[][n+1]){
    int flag = 1;
    for(int c=0;c<n;c++){
        for(int r=0;r<n-1;r++){
            if(str[r][c] > str[r+1][c]){
                flag = 0;
                break;
            }       
        }
        if (!flag) break;
    }
    if(flag == 1)
        printf("YES\n");
    else
        printf("NO\n");
}

int main(){
    int t;
    scanf("%d",&t);
    for(int i=0;i<t;i++){
        int n;
        scanf("%d",&n);
        char str[n][n+1];
        for(int i=0;i<n;i++){
            scanf(" %s",&str[i]);
            qsort(str[i],n,sizeof(char),compare);
        }
        grid(n,str);
    }
}
