// to find absolute difference between sum of primary diagonal and secondary diagonal

#include<stdio.h>
#include <stdlib.h>

void abs_diff(int n,int arr[][n]){
    int i,sum1=0,sum2=0;
    for(i=0;i<n;i++){
        sum1 += arr[i][i];
        sum2 += arr[i][n-1-i];
    }
    printf("%d",abs(sum1 - sum2));
}

int main()
{
    int n,i,j;
    scanf("%d",&n);
    int arr[n][n];
    for(i=0;i<n;i++){
        for(j=0;j<n;j++){
            scanf("%d",&arr[i][j]);
        }
    }
    abs_diff(n,arr);
    return 0;
}
