#include <stdio.h>

int birthday(int n,int arr[],int date,int mon){
    int i,j,count=0;
    for(i=0;i<=n-mon;i++){
        int sum = 0;
        for(j=0;j<mon;j++){
            sum += arr[i+j];
        }
        if(sum == date){
            count++;
        }
    }
    return count;
}

int main()
{
    int n;
    scanf("%d",&n);
    int arr[n],i;
    for(i=0;i<n;i++){
        scanf("%d",&arr[i]);
    }
    int date,mon;
    scanf("%d %d",&date,&mon);
    int count = birthday(n,arr,date,mon);
    printf("%d",count);
    return 0;
}
