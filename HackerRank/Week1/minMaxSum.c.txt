//minimum and maximum values that can be calculated by summing exactly four of the five integers.

#include <stdio.h>


// minimum sum and maximum sum
void minMaxSum(int arr[]){
    int total_sum = 0,i;
    int min = arr[0];
    int max = arr[0];
    for(i=0;i<5;i++){
        total_sum += arr[i];
        
        if(arr[i] < min)
            min = arr[i];
            
        if(arr[i] > max)
            max = arr[i];

    }
    printf("%lld ", total_sum - max);
    printf("%lld", total_sum - min);
}

int main(){
    int arr[5];
    for(int i=0;i<5;i++){
        scanf("%d",&arr[i]);    //array elements
    }
    minMaxSum(arr);
}
