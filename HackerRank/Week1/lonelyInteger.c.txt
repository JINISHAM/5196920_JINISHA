// in ana array of integers find uniq element
#include<stdio.h>

int lonelyInteger(int n,int arr[]){
    int i,result = 0;
    for(i=0;i<n;i++){
        result ^= arr[i];
    }
    return result;
}
int main()
{
    int n;
    scanf("%d",&n);
    int arr[n];
    for(int i=0;i<n;i++){
        scanf("%d",&arr[i]);
    }
    int result = lonelyInteger(n,arr);
    printf("%d",result);
    return 0;
}
