#include <stdio.h>

void countingSort(int n,int arr[]){
    int count[100] = {0};
    int i;
    for(i=0;i<n;i++){
        count[arr[i]]++;
    }
    for(i=0;i<100;i++){
        printf("%d ",count[i]);
    }
}

int main()
{
    int n,i;
    scanf("%d",&n);
    int arr[n];
    for(i=0;i<n;i++){
        scanf("%d",&arr[i]);
    }
    countingSort(n,arr);
    return 0;
}
