//Conversion of 12 hour format to military format

#include <stdio.h>
#include <string.h>

void military_time(int hour,int min,int sec,char period[]){
    if(strcmp(period,"AM") == 0){
        if(hour == 12)
            hour = 00;
    }
    else if(strcmp(period, "PM")== 0){
        if(hour !=12)
            hour += 12;
    }
    
    printf("%02d:%02d:%02d",hour,min,sec);
}
int main()
{
    char time[20];
    scanf("%s",time);
    int hour,min,sec;
    char period[3];
    //string scanf
    sscanf(time,"%2d:%2d:%2d%2s",&hour,&min,&sec,period);
    military_time(hour,min,sec,period);
    return 0;
}
