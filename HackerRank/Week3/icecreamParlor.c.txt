#include <stdio.h>

void icecreamParlor(int rs,int n,int flavours[]){
    for(int i=0;i<n;i++){
        for(int j=i+1;j<n;j++){
            if(flavours[i]+flavours[j] == rs){
                printf("%d %d\n",i+1,j+1);
            }
        }
    }
}

int main()
{
    int t;
    scanf("%d",&t);
    while(t--){
        int rs;
        scanf("%d",&rs);
        int n;
        scanf("%d",&n);
        int flavours[n];
        for(int i=0;i<n;i++){
            scanf("%d",&flavours[i]);
        }
        icecreamParlor(rs,n,flavours);
    }
    return 0;
}
