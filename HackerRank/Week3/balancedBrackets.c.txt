#include <stdio.h>
#include <string.h>
#define MAX 1000

char stack[MAX];
int top = -1;

void push(char c){
    if (top == MAX - 1) return;
    stack[++top] = c;
}

char pop(){
    if(top == -1)
        return '\0';
    else
        return stack[top--];
}

int ismatch(char open,char close){
    if(open == '(' && close == ')') return 1;
    if(open == '[' && close == ']') return 1;
    if(open == '{' && close == '}') return 1;
    return 0;
}

int isbalancedBrackets(char str[]){
    top = -1;
    char x;
    int len = strlen(str);
    for(int i=0;i<len;i++){
        if(str[i] == '(' || str[i] == '[' || str[i] == '{'){
            push(str[i]);
        }
        else if(str[i] == ')' || str[i] == ']' || str[i] == '}'){
            x = pop();
            if(!ismatch(x,str[i]))
                return 0;
        }
    }
    return(top == -1);
}

int main()
{
    int n;
    scanf("%d",&n);
    char str[MAX];
    while(n--){
        scanf("%s",str);
        if(isbalancedBrackets(str))
            printf("YES\n");
        else
            printf("NO\n");
    }
    return 0;
}
