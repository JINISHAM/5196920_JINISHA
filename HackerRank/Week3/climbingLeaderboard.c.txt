#include <stdio.h>

int removeDuplicates(int n,int ranked[]){
    int j=0;
    for(int i=1;i<n;i++){
        if(ranked[i] != ranked[j]){
            j++;
            ranked[j] = ranked[i];
        }
    }
    return j+1; 
}

void climbingLeaderboard(int n,int ranked[],int m,int players[]){
    n = removeDuplicates(n,ranked);
    int result[m];
    int i = n-1;
    for(int j=0;j<m;j++){
        while(i>=0 && players[j] >= ranked[i])
            i--;
            
        result[j] = i+2;    
    }
    
    for(int i=0;i<m;i++){
        printf("%d\n",result[i]);
    }
}

int main()
{
    int n;
    scanf("%d",&n);
    int ranked[n];
    for(int i=0;i<n;i++)
        scanf("%d",&ranked[i]);
        
    int m;
    scanf("%d",&m);
    int players[m];
    for(int i=0;i<m;i++)
        scanf("%d",&players[i]);
    
    climbingLeaderboard(n,ranked,m,players);
    return 0;
}
