#include <stdio.h>
#include <stdbool.h>

#define MAX 100000

bool isPrime(int x){
    if(x <= 1) return false;
    for(int i=2;i*i <= x;i++){
        if(x%i == 0)
            return false;
    }
    return true;
}

void waiter(int n,int arr[],int q){
    int answers[MAX];
    int ansindex = 0;
    int primes[q];
    int index = 0,num = 2;
    
    //generates 1st q primes
    while(index < q){
        if(isPrime(num)){
            primes[index] = num;
            index++;
        }
        num++;
    }
    
    int top = n-1;    //plate on top
    
    for(int i=0;i<q;i++){
        int stackA[MAX],stackB[MAX];
        int topA = -1,topB = -1;
        int prime = primes[i];
        
        while(top >= 0){
            int plate = arr[top--];
            if(plate % prime == 0)
                stackB[++topB] = plate;
            else
                stackA[++topA] = plate;
        }
        
        //stackB to answers arrray
        while(topB >= 0){
            answers[ansindex++] = stackB[topB--];
        }
        
        //treat stackA as new arr(A0)
        for (int j=0;j<=topA;j++) {
            arr[j] = stackA[j];
        }
        top = topA;
    }
    
    //remaining elements in stackA after q iterations
    while(top >= 0){
        answers[ansindex++] = arr[top--];
    }
    
    for(int i=0;i<ansindex;i++){
        printf("%d\n",answers[i]);
    }  
}

int main()
{
    int n,q;
    scanf("%d %d",&n,&q);
    int arr[n];
    for(int i=0;i<n;i++){
        scanf("%d",&arr[i]);
    }
    waiter(n,arr,q);
    return 0;
}
